// Função auxiliar para renderizar estrelas
const renderStars = (rating) => {
    // Escala baseada na notação de 10 corações/estrelas do Letterboxd (meia estrela = 0.5)
    const totalStars = 5;
    const fullStars = Math.floor(rating); 
    const isHalf = rating % 1 !== 0; 
    
    let stars = '★'.repeat(fullStars);

    if (isHalf) {
        stars += '½';
    }
    // Adiciona estrelas vazias
    stars += '☆'.repeat(totalStars - Math.ceil(rating));

    return stars;
};

            let arr = word.split('');

                // Troca o segundo e o penúltimo caractere
                let temp = arr[1];
                arr[1] = arr[arr.length - 2];
                arr[arr.length - 2] = temp;

                // Inverte para palavras curtas
